const ADVANCE = new Set(['New', 'Phoned - no answer']);
function logAction(l, type) {
  const t = todayISO();
  const patch = {};
  if ((type === 'call-spoke' || type === 'email') && ADVANCE.has(l.status)) patch.status = 'Replied - no quote';
  if (type === 'call-none' && l.status === 'New') patch.status = 'Phoned - no answer';
  if (type === 'drawing') patch.drawingSent = t;
  if (!l.nextFollowUp || l.nextFollowUp <= t) patch.nextFollowUp = addDays(t, type === 'call-none' ? 1 : 3);
  const kind = type === 'call-spoke' || type === 'call-none' ? 'call' : type;
  const action = { type: kind, from: l.status, to: patch.status || l.status };
  if (kind === 'call') action.outcome = type === 'call-none' ? 'no answer' : 'spoke';
  if (kind === 'drawing') action.rev = drawingCount(l) + 1;
  return save(l.id, patch, action);
}

function kindOf(e) {
  if (!e) return null;
  if (e.type === 'call' || e.type === 'email' || e.type === 'drawing' || e.type === 'chase') return e.type;
  if (e.type !== 'status') return null;
  if (e.to === 'Quote sent') return 'quote';
  if (e.to === 'Won') return 'won';
  if (e.to === 'Lost') return 'lost';
  if (e.to === 'Phoned - no answer') return 'call';
  if (e.to === 'Replied - no quote') return 'email';
  return null;
}

const who = (by) => (!by ? 'Automatic' : NAMES[String(by).toLowerCase()] || String(by).split('@')[0]);
function periodRange(p) {
  const t = todayISO();
  const d = new Date(t + 'T12:00:00');
  const mon = addDays(t, -((d.getDay() + 6) % 7));
  if (p === 'today') return [t, t];
  if (p === 'yesterday') return [addDays(t, -1), addDays(t, -1)];
  if (p === 'date') { const v = document.getElementById('period-date').value || t; return [v, v]; }
  if (p === 'week') return [mon, t];
  if (p === 'lastweek') return [addDays(mon, -7), addDays(mon, -1)];
  if (p === 'month') return [t.slice(0, 8) + '01', t];
  if (p === '30') return [addDays(t, -29), t];
  return ['0000-00-00', t];
}

function dueList() {
  const t = todayISO();
  const items = [];
  for (const l of leads) {
    if (!OPEN.has(l.status)) continue;
    if (l.nextFollowUp && l.nextFollowUp <= t) { items.push({ l, why: 'Follow-up due', when: l.nextFollowUp, late: l.nextFollowUp < t, rank: 0 }); continue; }

    if ((l.status === 'Quote sent' || l.status === 'Chased') && !l.nextFollowUp && last && last <= addDays(t, -7)) { items.push({ l, why: 'No chase for 7+ days', when: last, late: true, rank: 1 }); continue; }

      h('span', { class: 'c-id mono muted' }, l.id),
      h('span', { class: 'c-date num muted' }, prettyDate(l.date)),
      h('span', { class: 'name' }, l.name || '(no name)'),
      h('span', { class: 'enq' }, l.enquiry || l.postcode || ''),
      h('span', { class: 'c-status' }, pill(l.status)),
      h('span', { class: 'c-pri pri ' + (l.priority || '') }, l.priority || ''),
      h('span', { class: 'val' }, gbp(l.quote)),
      h('span', { class: 'meta-sm' }, [l.id, prettyDate(l.date), l.priority, gbp(l.quote)].filter(Boolean).join(' · ')));
    return h('div', { class: 'row' + (open ? ' open' : ''), role: 'listitem' }, head, open ? editor(l) : null);
  }));
}

function field(label, input, wide) { return h('label', { class: wide ? 'wide' : null }, label, input); }
function editor(l) {
  const dis = !canWrite;
  const sel = (opts, val, onchange, aria) => { const s = h('select', { disabled: dis, 'aria-label': aria, onchange }); opts.forEach((o) => { const v = typeof o === 'string' ? o : o.k; const t = typeof o === 'string' ? (o || '-') : o.label; const op = h('option', { value: v }, t); if (v === (val || '')) op.selected = true; s.append(op); }); return s; };
  const blurSave = (key, parse) => (e) => { const v = parse ? parse(e.target.value) : e.target.value.trim(); if ((l[key] ?? '') === (v ?? '')) return; save(l.id, { [key]: v }, { type: 'edit', field: key }); };
  const facts = h('dl', { class: 'facts' },
    h('dt', {}, 'Source'), h('dd', {}, l.source || ''),
    h('dt', {}, 'Phone'), h('dd', { class: 'num' }, l.phone || ''),
    h('dt', {}, 'Email'), h('dd', {}, l.email || ''),
    h('dt', {}, 'Area'), h('dd', {}, l.postcode || ''),
    h('dt', {}, 'Enquiry'), h('dd', {}, l.enquiry || ''),
    h('dt', {}, 'Drawing sent'), h('dd', {}, drawingCount(l) ? `${drawingCount(l)} time${drawingCount(l) > 1 ? 's' : ''}, last ${prettyDate(l.drawingSent)}${drawingCount(l) > 1 ? ' (rev ' + drawingCount(l) + ')' : ''}` : (l.drawingSent ? prettyDate(l.drawingSent) : '–')),
    h('dt', {}, 'Chases'), h('dd', { class: 'num' }, String(l.chases || 0) + (l.lastChased ? ` (last ${prettyDate(l.lastChased)})` : '')));
  const ask = (label, fn) => { ui.confirm = { id: l.id, label, fn }; renderList(); };
  const quickBtn = (txt, fn, primary) => h('button', { class: 'btn' + (primary ? ' primary' : ''), type: 'button', disabled: dis, onclick: () => ask(txt, fn) }, txt);
  const pending = ui.confirm && ui.confirm.id === l.id ? ui.confirm : null;
  const confirmBar = pending ? h('div', { class: 'confirm', role: 'alertdialog', 'aria-label': 'Confirm' },
    h('span', {}, 'Log ', h('b', {}, pending.label), ' for ', h('b', {}, l.name || l.id), '?'),
    h('button', { class: 'btn primary', type: 'button', onclick: () => { const f = pending.fn; ui.confirm = null; f(); } }, 'Confirm'),
    h('button', { class: 'btn', type: 'button', onclick: () => { ui.confirm = null; renderList(); } }, 'Cancel')) : null;
  const form = h('div', { class: 'form' },
    confirmBar,
    pending ? null : h('div', { class: 'quick' },
      quickBtn('Called – no answer', () => logAction(l, 'call-none')),
      quickBtn('Called – spoke', () => logAction(l, 'call-spoke')),
      quickBtn('Emailed', () => logAction(l, 'email')),
      quickBtn(drawingCount(l) ? `Revised drawing sent (rev ${drawingCount(l) + 1})` : 'Drawing sent', () => logAction(l, 'drawing')),
      quickBtn('Quote sent', () => setStatus(l, 'Quote sent', { nextFollowUp: l.nextFollowUp || addDays(todayISO(), 3) }), true),
      quickBtn('Chased today', () => chase(l)),
      quickBtn('Won', () => setStatus(l, 'Won', { nextFollowUp: '' })),
      quickBtn('Lost', () => setStatus(l, 'Lost', { nextFollowUp: '' }))),
    field('Status', sel(STATUSES, l.status, (e) => { const to = e.target.value; ask('status: ' + ((SBY[to] || {}).label || to), () => setStatus(l, to)); }, 'Status')),
    field('Owner', sel(OWNERS, l.owner, (e) => save(l.id, { owner: e.target.value }, { type: 'edit', field: 'owner' }), 'Owner')),
    field('Quote £ (ex VAT)', h('input', { type: 'number', min: '0', step: '1', inputmode: 'numeric', value: l.quote || '', disabled: dis, onchange: blurSave('quote', (v) => (v === '' ? null : Number(v))) })),
    field('Next follow-up', h('input', { type: 'date', value: l.nextFollowUp || '', disabled: dis, onchange: blurSave('nextFollowUp') })),
    field('Priority', sel(['High', 'Medium', 'Low'], l.priority, (e) => save(l.id, { priority: e.target.value }, { type: 'edit', field: 'priority' }), 'Priority')),
    l.status === 'Lost' ? field('Lost reason', sel(LOST, l.lostReason, (e) => save(l.id, { lostReason: e.target.value }, { type: 'edit', field: 'lostReason' }), 'Lost reason')) : h('span'),
    field('Notes', h('textarea', { disabled: dis, onchange: blurSave('notes') }, l.notes || ''), true),
    h('ul', { class: 'hist' }, ...(l.history || []).slice().reverse().slice(0, 12).map((e) => h('li', {}, `${new Date(e.at).toLocaleString('en-GB', { day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' })}${e.by ? ' (' + String(e.by).split('@')[0] + ')' : ''}: ${e.type === 'call' ? (e.outcome === 'no answer' ? 'Called, no answer' : 'Called, spoke') : e.type === 'email' ? 'Emailed' : e.type === 'drawing' ? ((e.rev || 1) > 1 ? 'Revised drawing sent (rev ' + e.rev + ')' : 'Drawing sent') : e.type === 'invoice' ? 'Invoice sent: ' + (e.stage || '') : e.type === 'paid' ? 'Payment received: ' + (e.stage || '') : e.type === 'chase' ? 'Chased' : e.type === 'status' ? `${(SBY[e.from] || {}).label || e.from || '-'} → ${(SBY[e.to] || {}).label || e.to}` : e.type === 'created' ? 'Lead added' : e.type === 'import' ? 'Imported from the lead review' : e.type === 'wix' ? 'Website form' : 'Edited ' + (e.field || '')}`))));
  const showPay = l.status === 'Won' || (Array.isArray(l.payments) && l.payments.length);
  return h('div', { class: 'edit' }, facts, form, showPay ? paymentsBlock(l, ask) : null);
}
